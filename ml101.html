# 1. Data
import torch
x = torch.linspace(-4, 4, 25)
y = 2 * x + 1 + 0.8 * torch.randn(25)

# 2. Model — just two numbers
w = torch.randn(1, requires_grad=True)
b = torch.randn(1, requires_grad=True)

# 3. Optimizer
opt = torch.optim.SGD([w, b], lr=0.05)

# 4. Train
for step in range(200):
    y_pred = w * x + b
    loss = ((y_pred - y) ** 2).mean()
    opt.zero_grad()
    loss.backward()
    opt.step()

print(f"w={w.item():.2f}, b={b.item():.2f}")
